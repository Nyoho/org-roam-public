#+BEGIN_SRC sh
systemctl status service-name
#+END_SRC

有効化する
#+BEGIN_SRC sh
systemctl enable /etc/systemd/system/hogehoge.service
#+END_SRC

enable/disabled を確認する
#+begin_src sh
systemctl list-unit-files
#+end_src

hogehoge.service がうまく動かないときに

#+begin_src sh
systemd-analyze verify hogehoge.service
#+end_src
